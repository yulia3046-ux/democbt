Koneksi Google Sheets & Apps Script

Otomatisasi pencatatan nilai ujian siswa secara real-time ke Spreadsheet.

Kode Google Apps Script (Mendukung Skor + Live Kamera CCTV + Gambar Asli di Sheets): 
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var data = JSON.parse(e.postData.contents);

  // 1. Jika paket kiriman adalah Tangkap Kamera Live (CCTV Snapshot)
  if (data.action === "camera_snapshot") {
    var camSheet = ss.getSheetByName("LIVE_CAMERA") || ss.insertSheet("LIVE_CAMERA");
    if (camSheet.getLastRow() === 0) {
      camSheet.appendRow(["NISN", "Nama", "Kelas", "Mapel", "Progres", "Pelanggaran", "WaktuUpdate", "FotoBase64", "Visual Foto Kamera"]);
      camSheet.setColumnWidth(9, 180);
    }
    var rows = camSheet.getDataRange().getValues();
    var foundRow = -1;
    for (var i = 1; i < rows.length; i++) {
      if (String(rows[i][0]) === String(data.nisn)) {
        foundRow = i + 1;
        break;
      }
    }
    var rowData = [data.nisn, data.name, data.className, data.subject, data.progress, data.violations, data.timestamp, data.snapshot];
    var targetRow = foundRow > 0 ? foundRow : camSheet.getLastRow() + 1;

    camSheet.getRange(targetRow, 1, 1, 8).setValues([rowData]);
    camSheet.setRowHeight(targetRow, 110);

    // Render Base64 menjadi Gambar Asli langsung di dalam sel Google Sheets (Kolom ke-9)
    try {
      var imgCell = SpreadsheetApp.newCellImage().setSourceUrl(data.snapshot).build();
      camSheet.getRange(targetRow, 9).setValue(imgCell);
    } catch (err) {}

    return ContentService.createTextOutput(JSON.stringify({status: "cam_updated"})).setMimeType(ContentService.MimeType.JSON);
  }

  // 2. Jika paket kiriman adalah Nilai Akhir Ujian Siswa
  var sheet = ss.getSheets()[0];
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Waktu", "Nama Siswa", "NISN", "Kelas", "Mata Pelajaran", "Benar", "Total Soal", "Skor Akhir", "Pelanggaran"]);
  }
  sheet.appendRow([data.timestamp, data.name, data.nisn, data.className, data.subject, data.correct, data.total, data.score, data.violations]);
  return ContentService.createTextOutput(JSON.stringify({status: "score_saved"})).setMimeType(ContentService.MimeType.JSON);
}

// Endpoint GET untuk diambil oleh Dashboard Live CCTV Pengawas
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var camSheet = ss.getSheetByName("LIVE_CAMERA");
  if (!camSheet || camSheet.getLastRow() <= 1) {
    return ContentService.createTextOutput(JSON.stringify({feeds: []})).setMimeType(ContentService.MimeType.JSON);
  }
  var values = camSheet.getDataRange().getValues();
  var feeds = [];
  for (var i = 1; i < values.length; i++) {
    feeds.push({
      nisn: values[i][0],
      name: values[i][1],
      className: values[i][2],
      subject: values[i][3],
      progress: values[i][4],
      violations: values[i][5],
      timestamp: values[i][6],
      snapshot: values[i][7]
    });
  }
  return ContentService.createTextOutput(JSON.stringify({feeds: feeds})).setMimeType(ContentService.MimeType.JSON);
}
